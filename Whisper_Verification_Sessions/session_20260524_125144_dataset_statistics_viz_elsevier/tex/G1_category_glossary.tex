\begin{table}[!t]
  \centering
  \caption{Glossary of the 11 Indonesian sentence-type categories
           used in the corpus.}
  \label{tab:category_glossary}
  \small
  \begin{tabular}{lll}
  \toprule
  Indonesian label    & English gloss        & Function \\
  \midrule
  Kalimat\_Declarative  & Declarative          & Statement that asserts a fact \\
  Kalimat\_Clarification & Clarification        & Request to clarify or rephrase \\
  Kalimat\_Conditional & Conditional          & If--then construction \\
  Kalimat\_Confirmation  & Confirmation         & Yes/no confirmation request \\
  Kalimat\_Negation      & Negation             & Negated assertion (\emph{tidak / bukan}) \\
  Kalimat\_Scheduling & Scheduling           & Time-related plan or appointment \\
  Kalimat\_Imperative    & Command / Imperative & Direct instruction (often telegraphic) \\
  Kalimat\_Persuasive   & Persuasive           & Multi-clause argumentation, longest type \\
  Kalimat\_Rhetorical     & Rhetorical           & Question whose answer is implied \\
  Kalimat\_Exclamatory      & Exclamation          & Surprise / emphasis \\
  Kalimat\_Interrogative       & Interrogative        & Genuine information-seeking question \\
  \bottomrule
  \end{tabular}
